model users {
  id                  Int                   @id @default(autoincrement())
  password            String                @db.VarChar(255)
  name                String?               @db.VarChar(255)
  avatar              String?               @db.VarChar(255)
  email               String                @unique(map: "users_email_unique") @db.VarChar(255)
  mobile              String?               @unique(map: "users_phone_unique") @db.VarChar(255)
  logout_at           DateTime?             @default(now())
  created_at          DateTime?             @default(now())
  updated_at          DateTime?             @default(now())
  deleted_at          DateTime?


  roles               user_role[]
  make_codes          make_code_user[]
  jobs                jobs[]
  vehicle_incidents   vehicle_incidents[]
  production_tracking production_tracking[]
  block_user          block_items[]         @relation("block_user")
  unblock_user        block_items[]         @relation("unblock_user")
}

model make_codes {
  id         Int              @id @default(autoincrement())
  reference  String           @unique @db.VarChar(255)
  title      String           @db.VarChar(255)
  created_at DateTime?        @default(now())
  updated_at DateTime?        @default(now())
  deleted_at DateTime?
  users      make_code_user[]
}

model car_models {
  id                 Int                @id @default(autoincrement())
  reference          String             @unique @db.VarChar(255)
  title              String?            @db.VarChar(255)
  created_at         DateTime?          @default(now())
  updated_at         DateTime?          @default(now())
  deleted_at         DateTime?
  production_line    production_lines?  @relation(fields: [production_line_id], references: [id])
  production_line_id Int?
  incident_sources   incident_sources[]
}

model vehicles {
  vin                     String                @id @default(uuid())
  production_order_number String                @unique
  engine_no               String?
  cabin_no                String?
  car_model_id            Int?
  make_code_id            Int?
  created_at              DateTime?             @default(now())
  updated_at              DateTime?             @default(now())
  deleted_at              DateTime?
  vehicle_incidents       vehicle_incidents[]
  production_trackings    production_tracking[]
  block_items             block_items[]
}

model roles {
  id          Int               @id @default(autoincrement())
  title       String            @db.VarChar(255)
  created_at  DateTime?         @default(now())
  updated_at  DateTime?         @default(now())
  deleted_at  DateTime?
  users       user_role[]
  permissions role_permission[]
}

model permissions {
  id         Int               @id @default(autoincrement())
  name       String            @unique @db.VarChar(255)
  title      String            @db.VarChar(255)
  path       String            @db.VarChar(255)
  method     String            @db.VarChar(255)
  created_at DateTime?         @default(now())
  updated_at DateTime?         @default(now())
  deleted_at DateTime?
  roles      role_permission[]
}

model user_role {
  id      Int    @id @default(autoincrement())
  user    users? @relation(fields: [user_id], references: [id])
  role    roles? @relation(fields: [role_id], references: [id])
  user_id Int?
  role_id Int?
}

model role_permission {
  id            Int          @id @default(autoincrement())
  permission    permissions? @relation(fields: [permission_id], references: [id])
  role          roles?       @relation(fields: [role_id], references: [id])
  permission_id Int?
  role_id       Int?
}

model make_code_user {
  id           Int         @id @default(autoincrement())
  user         users?      @relation(fields: [user_id], references: [id])
  make_code    make_codes? @relation(fields: [make_code_id], references: [id])
  user_id      Int?
  make_code_id Int?
}

model jobs {
  id         Int       @id @default(autoincrement())
  hash       String    @unique
  input      Json
  output     Json?
  status     String?
  queueName  String
  users      users     @relation(fields: [user_id], references: [id])
  created_at DateTime? @default(now())
  updated_at DateTime? @default(now())
  deleted_at DateTime?
  user_id    Int
}

model vehicle_incidents {

  user_id                Int

model incident_sources {
  id           Int        @id @default(autoincrement())
  car_model_id Int

  element      elements   @relation(fields: [element_id], references: [id])
  incident     incidents  @relation(fields: [incident_id], references: [id])
  car_model    car_models @relation(fields: [car_model_id], references: [id])

  updated_at   DateTime?  @default(now())
  deleted_at   DateTime?
}

model production_tracking {
  id                Int                 @id @default(autoincrement())
  model_type        String // it's about which physical section of the industry
  model_id          Int // this Id represents the physical section of the identity
  user_id           Int
  user              users               @relation(fields: [user_id], references: [id])
  created_at        DateTime?           @default(now())
  updated_at        DateTime?           @default(now())
  deleted_at        DateTime?
  vehicle_incidents vehicle_incidents[]
  vehicle           vehicles?           @relation(fields: [vin], references: [vin])
  vin               String?
  type              String              @default("enter")
}

model block_reasons {
  id          Int           @id @default(autoincrement())

}

model block_items {
  id              Int           @id @default(autoincrement())
  block_user      users         @relation("block_user", fields: [blocked_by], references: [id])
  blocked_by      Int
  unblock_user    users?        @relation("unblock_user", fields: [unblocked_by], references: [id])
  unblocked_by    Int?
  block_reason    block_reasons @relation(fields: [block_reason_id], references: [id])
  block_reason_id Int
  unblocked_at    DateTime?
  description     String?
  created_at      DateTime?     @default(now())
  updated_at      DateTime?     @default(now())
  deleted_at      DateTime?
  vehicle         vehicles?     @relation(fields: [vin], references: [vin])
  vin             String
}
